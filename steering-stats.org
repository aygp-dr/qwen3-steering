* Preamble

Burke's terministic screen, measured without asking the model anything.

The three conditions — =normal=, =terse=, =verbose= — produce distributions
over observable surface features: word count, sentence count, type-token ratio,
mean word length, punctuation density. If steering is real, these distributions
are separable. K-means should recover the three classes from features alone.
If it can't, the steering isn't doing what we think.

This file contains no Ollama calls. The synthetic data is parameterised by
empirically observed medians and standard deviations from =actadd.py= runs.
Replace =OBSERVED_STATS= with real measurements and every downstream analysis
updates automatically.

** Observed Empirical Baselines

From =C-27= (raw vec, α=2.0, layer 12, mutex prompt, n=30 runs each):

| Condition | Median words | Std | Median sentences | Std |
|-----------+--------------+-----+------------------+-----|
| normal    |           82 |  18 |                6 | 1.8 |
| terse     |           16 |   6 |                2 | 0.9 |
| verbose   |          194 |  41 |               14 | 3.2 |

These are the ground truth. The analysis below asks: are these distributions
distinguishable by a classifier that has never seen the prompts?

#+begin_src python
# ── Empirical parameters (from C-27 runs) ─────────────────────────────────
OBSERVED_STATS = {
    #          median  std   sent_med sent_std  ttr_med ttr_std  wlen_med wlen_std
    "normal":  ( 82,   18,     6,      1.8,     0.68,   0.05,    4.8,    0.3 ),
    "terse":   ( 16,    6,     2,      0.9,     0.82,   0.04,    5.1,    0.4 ),
    "verbose": (194,   41,    14,      3.2,     0.58,   0.06,    4.5,    0.4 ),
}

N_SAMPLES = 120   # 40 per condition — realistic for n=30..50 steering runs

def generate_condition(name, n=40):
    med_w, std_w, med_s, std_s, med_t, std_t, med_l, std_l = OBSERVED_STATS[name]
    # word count: log-normal to avoid negative values
    words = np.random.lognormal(
        mean=np.log(med_w), sigma=std_w / med_w, size=n
    ).clip(1)
    # sentence count: correlated with words
    sents = np.random.lognormal(
        mean=np.log(med_s), sigma=std_s / med_s, size=n
    ).clip(1)
    # type-token ratio: beta-distributed (bounded 0–1)
    a, b = med_t * 10, (1 - med_t) * 10
    ttr = np.random.beta(a, b, size=n)
    # mean word length: normal
    wlen = np.random.normal(med_l, std_l, size=n).clip(2)
    # punctuation density: inversely correlated with TTR, weakly
    punct = np.random.normal(0.08 - (ttr - 0.68) * 0.05, 0.02, size=n).clip(0)
    return pd.DataFrame({
        "words": words, "sentences": sents,
        "ttr": ttr, "mean_word_len": wlen,
        "punct_density": punct,
        "condition": name,
        "label": {"normal": 0, "terse": 1, "verbose": 2}[name],
    })

df = pd.concat([
    generate_condition("normal",  n=N_SAMPLES // 3),
    generate_condition("terse",   n=N_SAMPLES // 3),
    generate_condition("verbose", n=N_SAMPLES // 3),
], ignore_index=True)

FEATURES = ["words", "sentences", "ttr", "mean_word_len", "punct_density"]
PALETTE   = {"normal": "#4C72B0", "terse": "#DD8452", "verbose": "#55A868"}
LABELS    = ["normal", "terse", "verbose"]

print(df.groupby("condition")[FEATURES].agg(["median", "std"]).round(2).to_string())

** PCA Projection with Cluster Ellipses

#+begin_src python
pca = PCA(n_components=2, random_state=SEED)
X2  = pca.fit_transform(X)
df["pc1"], df["pc2"] = X2[:, 0], X2[:, 1]
ev = pca.explained_variance_ratio_

def confidence_ellipse(x, y, ax, n_std=2.0, **kwargs):
    cov = np.cov(x, y)
    pearson = cov[0, 1] / np.sqrt(cov[0, 0] * cov[1, 1])
    rx, ry = np.sqrt(1 + pearson), np.sqrt(1 - pearson)
    ellipse = Ellipse((0, 0), width=rx * 2, height=ry * 2, **kwargs)
    scale_x = np.sqrt(cov[0, 0]) * n_std
    scale_y = np.sqrt(cov[1, 1]) * n_std
    t = transforms.Affine2D() \
        .rotate_deg(45) \
        .scale(scale_x, scale_y) \
        .translate(np.mean(x), np.mean(y))
    ellipse.set_transform(t + ax.transData)
    return ax.add_patch(ellipse)

fig, axes = plt.subplots(1, 2, figsize=(13, 5))
fig.suptitle(
    f"PCA Projection  (PC1={ev[0]:.1%}, PC2={ev[1]:.1%} variance)\n"

    fontsize=10

* Statistical Tests
:PROPERTIES:
:header-args:python: :tangle stats/tests.py :mkdirp t
:END:

Kruskal-Wallis (non-parametric, appropriate for non-normal word count
distributions) across all three conditions, then pairwise Mann-Whitney U
with Bonferroni correction. This is the table that goes in a paper.

#+begin_src python
from itertools import combinations
from scipy.stats import kruskal, mannwhitneyu

rows = []
for feat in FEATURES:
    groups = [df[df.condition == c][feat].values for c in LABELS]
    H, p_kw = kruskal(*groups)
    rows.append({
        "feature": feat, "H": round(H, 2), "p_kruskal": p_kw,
        "significant": p_kw < 0.05
    })

kw_df = pd.DataFrame(rows)
print("=== Kruskal-Wallis (H₀: identical distributions across conditions) ===")
print(kw_df.to_markdown(index=False))

# Pairwise Mann-Whitney with Bonferroni
pairs = list(combinations(LABELS, 2))
n_tests = len(FEATURES) * len(pairs)
alpha_corrected = 0.05 / n_tests

mw_rows = []
for feat in FEATURES:
    for c1, c2 in pairs:
        g1 = df[df.condition == c1][feat].values
        g2 = df[df.condition == c2][feat].values
        U, p = mannwhitneyu(g1, g2, alternative="two-sided")
        r = 1 - (2 * U) / (len(g1) * len(g2))   # rank-biserial correlation
        mw_rows.append({
            "feature": feat, "pair": f"{c1} vs {c2}",
            "U": round(U, 0), "p": round(p, 4),
            "effect_r": round(r, 3),
            "sig_bonf": p < alpha_corrected
        })

mw_df = pd.DataFrame(mw_rows)
print(f"\n=== Pairwise Mann-Whitney U (Bonferroni α={alpha_corrected:.4f}) ===")
print(mw_df.to_markdown(index=False))
#+end_src

* Effect Size Heatmap
:PROPERTIES:
:header-args:python: :tangle stats/effect_sizes.py :mkdirp t
:END:

Rank-biserial correlation =r= as a matrix: features × condition pairs.
Cells near ±1.0 are the features that most cleanly separate that pair.
This tells you which lens features to keep and which are noise.

#+begin_src python
pairs_display = ["normal\nvs terse", "normal\nvs verbose", "terse\nvs verbose"]
pair_keys     = [("normal","terse"), ("normal","verbose"), ("terse","verbose")]

effect_matrix = np.zeros((len(FEATURES), len(pair_keys)))
sig_matrix    = np.zeros_like(effect_matrix, dtype=bool)

for j, (c1, c2) in enumerate(pair_keys):
    for i, feat in enumerate(FEATURES):
        g1 = df[df.condition == c1][feat].values
        g2 = df[df.condition == c2][feat].values
        U, p = mannwhitneyu(g1, g2, alternative="two-sided")
        effect_matrix[i, j] = 1 - (2 * U) / (len(g1) * len(g2))
        sig_matrix[i, j] = p < alpha_corrected

fig, ax = plt.subplots(figsize=(7, 5))
im = ax.imshow(effect_matrix, cmap="RdBu", vmin=-1, vmax=1, aspect="auto")
plt.colorbar(im, ax=ax, label="rank-biserial r  (±1 = perfect separation)")

for i in range(len(FEATURES)):
    for j in range(len(pair_keys)):
        val = effect_matrix[i, j]
        marker = "★" if sig_matrix[i, j] else ""
        ax.text(j, i, f"{val:.2f}{marker}", ha="center", va="center",
                fontsize=9, color="white" if abs(val) > 0.6 else "black")

ax.set_xticks(range(len(pairs_display)))
ax.set_xticklabels(pairs_display, fontsize=9)
ax.set_yticks(range(len(FEATURES)))
ax.set_yticklabels([f.replace("_", " ") for f in FEATURES], fontsize=9)
ax.set_title(
    "Effect Size (rank-biserial r) — Feature × Condition Pair\n"
    "★ = significant after Bonferroni correction",

)
